//@version=5
indicator("CDC ActionZone V3 2020", overlay=true)
xsrc = input.source(close, "Source")
xprd1 = input.int(12, "Fast EMA Period")
xprd2 = input.int(26, "Slow EMA Period")
xsmooth = input.int(1, "Smoothing Period")

xPrice = ta.ema(xsrc, xsmooth)
FastMA = ta.ema(xPrice, xprd1)
SlowMA = ta.ema(xPrice, xprd2)

Bull = FastMA > SlowMA
Bear = FastMA < SlowMA

Green  = Bull and xPrice > FastMA
Blue   = Bear and xPrice > FastMA and xPrice > SlowMA
LBlue  = Bear and xPrice > FastMA and xPrice < SlowMA
Red    = Bear and xPrice < FastMA
Orange = Bull and xPrice < FastMA and xPrice < SlowMA
Yellow = Bull and xPrice < FastMA and xPrice > SlowMA

bColor = Green ? color.lime : Blue ? color.blue : LBlue ? color.aqua : Red ? color.red : Orange ? color.orange : Yellow ? color.yellow : color.black
barcolor(bColor)
slPct = input.float(3.0, "% Stop Loss")
rrRatio = input.float(2.0, "Risk:Reward Ratio (TP)")
isLong = input.bool(true, "Direction: true=Long, false=Short")

tpPrice = isLong ? close * (1 + (slPct * rrRatio) / 100) : close * (1 - (slPct * rrRatio) / 100)
slPrice = isLong ? close * (1 - slPct / 100) : close * (1 + slPct / 100)

plot(FastMA, "Fast EMA(12)", color=color.red, linewidth=1)
plot(SlowMA, "Slow EMA(26)", color=color.blue, linewidth=2)
plot(tpPrice, "Take Profit Line", color=color.new(color.lime, 20), style=plot.style_linebr, linewidth=2)
plot(slPrice, "Stop Loss Line", color=color.new(color.red, 20), style=plot.style_linebr, linewidth=2)
plotshape(Bull and Bear[1], "Buy (เขียวแรก)", shape.labelup, location.belowbar, color.lime, text="BUY", textcolor=color.black)
plotshape(Bear and Bull[1], "Sell (แดงแรก)", shape.labeldown, location.abovebar, color.red, text="SELL", textcolor=color.white)

ระบบอ่านกราฟแท่งเทียนตามกฎ Price Action (Candlestick Pattern & Wick Scanner)

BULLISH CANDLE SIGNAL

วิเคราะห์สัดส่วนเนื้อเทียน (Body %), ไส้เทียนบน-ล่าง (Wick Rejection), ตำแหน่งเกิดสัญญาณ (Context) และตรวจจับรูปแบบแท่งเทียน 1-3 แท่งอัตโนมัติ

1. รูปแบบแท่งเทียนล่าสุด 3-Bar Pattern
Bullish Engulfing (กลืนกินขาขึ้น)
แท่งเขียวใหญ่กลืนกินแท่งแดงก่อนหน้าเต็มแท่ง แรงซื้อชนะขาดลอย
2. ขนาดเนื้อเทียน (Body Ratio) Body: 68.5%
เนื้อเทียนแน่น (Strong Body)
ราคาปิดใกล้จุดสูงสุดของแท่ง สะท้อนแรงซื้อคุมตลาดจนจบแท่ง
3. การปฏิเสธราคา (Wick Ratio) บน 12% | ล่าง 19%
มีแรงรับซื้อจากไส้เทียนด้านล่าง
ไส้ล่างยาวกว่าไส้บน บ่งชี้มีแรงซื้อดักรับเมื่อราคาย่อตัวลง
4. ตำแหน่งเกิด & จุดคัทปลายไส้ Volume ยืนยัน
SL ปลายไส้: 77,380.0 USDT
เกิดใกล้โซนแนวรับ EMA / Fibo วาง SL ใต้หางแท่งเทียนปลอดภัย
ตารางผ่าโครงสร้าง 3 แท่งเทียนล่าสุด (OHLC + สัดส่วนเนื้อเทียน & ไส้เทียน + ความหมายตามกฎแท่งเทียน) อัปเดตอัตโนมัติตามคู่เหรียญและ Timeframe ที่เลือก
ลำดับแท่ง Open → Close Low ↔ High เนื้อเทียน (Body) ไส้บน / ไส้ล่าง การอ่านค่าตามกฎแท่งเทียน
สรุปกฎเหล็กการอ่านกราฟแท่งเทียน (4 เสาหลักที่ระบบใช้ประมวลผล)
กฎข้อที่ 1: ขนาดและสีของเนื้อเทียน (Real Body = พลังของเทรนด์)
• เนื้อเทียนยาวแน่น (Marubozu / Body > 65%): แรงซื้อหรือแรงขายคุมเกมเบ็ดเสร็จตั้งแต่วินาทีเปิดจนปิดแท่ง มักเกิดต้นคลื่น Wave 3 หรือจังหวะ Breakout
• เนื้อเทียนสั้น/เล็ก (Spinning Top / Doji < 15%): ตลาดลังเล แรงซื้อและแรงขายพอกัน หากเกิดหลังวิ่งมาไกลเป็นสัญญาณเตือนการพักตัวหรือกลับตัว
กฎข้อที่ 2: ความยาวไส้เทียน (Wicks / Shadows = การปฏิเสธราคา)
• ไส้ล่างยาวเกิน 2 เท่าของเนื้อเทียน (Hammer / Bullish Pin Bar): ราคาถูกทุบลงไปแต่มีแรงซื้อสวนกลับขึ้นมาปิดด้านบน = ปฏิเสธการลง (Bullish Rejection)
• ไส้บนยาวเกิน 2 เท่าของเนื้อเทียน (Shooting Star / Bearish Pin Bar): ราคาถูกไล่ขึ้นไปแต่โดนเทขายสวนลงมาปิดด้านล่าง = ปฏิเสธการขึ้น (Bearish Rejection)
กฎข้อที่ 3: รูปแบบแท่งเทียนคู่และ 3 แท่ง (Multi-Candle Structure)
• Engulfing (กลืนกิน): แท่งปัจจุบันมีเนื้อเทียนกลืนกินเนื้อเทียนก่อนหน้ามิดทั้งแท่ง ยืนยันการเปลี่ยนมือของผู้คุมตลาดทันที
• Morning Star / Evening Star (3 แท่งกลับตัว): แท่งยาวตามเทรนด์เดิม → แท่งเล็กพักตัว (Doji/Star) → แท่งยาวสวนกลับเกิน 50% ของแท่งแรก
กฎข้อที่ 4: ตำแหน่งเกิดสัญญาณ + Volume + การวาง Stop Loss
• แท่งเทียนกลับตัวจะมีความแม่นยำสูงก็ต่อเมื่อ เกิดที่แนวรับ-แนวต้านสำคัญ (EMA 12/26, แนวรับ Fibonacci ของ Elliott Wave) และมี OBV / Volume หนุน
• กฎการวาง Stop Loss: สำหรับฝั่ง Long ให้วาง SL ต่ำกว่าปลายไส้ล่างของแท่งสัญญาณเล็กน้อย และฝั่ง Short ให้วาง SL เหนือปลายไส้บนของแท่งสัญญาณ

4. จุดตัดขาดทุน (Stop Loss) 75,480.6 USDT (-3.00%) อยู่ต่ำกว่าทั้งเส้น CDC Slow EMA(26) (76,980.0) และต่ำกว่าฐานย่อ Wave 2 (76,500.0) ไม่โดนเจ้ามือสะบัดกิน SL ง่ายๆ
5. เป้าหมายทำกำไร 1 (TP1) 80,824.5 USDT (Fibo 127.2% | แบ่งปิด 50% = 0.0214 BTC) แบ่งปิดทำกำไรครึ่งแรกเพื่อล็อกเงินเข้ากระเป๋าก่อน แล้วเลื่อน SL ของครึ่งที่เหลือมาบังทุน (Break-Even = Free Trade)
6. เป้าหมายทำกำไรหลัก (TP2) 82,483.9 USDT (Fibo 161.8% / RR 1:2 | รับกำไร +200.00 USDT) เป้าหมายหลักของคลื่น Wave 3 หรือถือรันเทรนด์ต่อด้วย Trailing Stop (เส้น Slow EMA 26) จนกว่า CDC จะเปลี่ยนเป็นสีแดง
ส่วนที่ 3: ถ้าไม่เป็นไปตามแผน ต้องทำอย่างไรต่อ? (แผนฉุกเฉิน 3 สถานการณ์ — รู้ล่วงหน้าว่าต้องทำอะไรโดยไม่ใช้อารมณ์) กดปุ่มด้านล่างในแต่ละสถานการณ์เพื่อปรับค่าตารางเทรดตามแผนรับมือได้ทันที
กรณีที่ 1: ราคาไม่ไปต่อ ย่อเกิดแท่ง "สีเหลือง/สีส้ม" หรือปิด "สีแดง" Plan B1

• อาการบนกราฟ: ราคาหลุดเส้น Fast EMA(12) (77,510.0) แท่งเทียนเปลี่ยนเป็น สีเหลือง (Yellow) แต่ยังไม่หลุดเส้น Slow EMA(26) (76,980.0)
• วิธีรับมือเมื่อเป็นสีเหลือง: อย่าเพิ่งตกใจปิดมั่ว! ตามกฎ CDC สีเหลืองในขาขึ้นคือ "การย่อพักฐาน (Pullback)" ตราบใดที่ยังไม่หลุดฐาน Wave 2 (76,500.0) ให้ถือตามแผนเดิม
• แต่ถ้าแท่งเทียนปิดเป็น "สีแดง (Red)" + EMA12 ตัด EMA26 ลง + OBV หลุด EMA20: แปลว่ารอบขาขึ้นเสียทรงแล้ว ให้ คัดลอกปิดสถานะ (Manual Cut Loss) ที่เส้น Slow EMA26 ทันที เพื่อประหยัดความเสี่ยงให้เสียน้อยกว่า 1R (เสียเพียง ~0.4R - 0.5R)

กรณีที่ 2: ราคาทุบแรงจนชน Stop Loss (ผิดแผน 100% / Wave โมฆะ) Plan B2

• อาการบนกราฟ: ราคาทุบชน SL ที่ 75,480.6 และหลุดฐานคลื่น แปลว่า การนับคลื่นขึ้นชุดนี้เป็นโมฆะ (Wave Invalidation)
• กฎเหล็ก Money Management:
1. ยอมรับการคัททันที ห้ามขยับหนี SL และห้ามถัวเฉลี่ยขาลง (No Martingale): คุณเสียแค่ 100.00 USDT (2%) พอร์ตยังเหลือเงินสดอีก 4,900.00 USDT และเหลือกระสุนอีกถึง 49 นัด
2. ถอยออกมาตั้งหลัก & พลิกหน้าเทรดอย่างมีระบบ: สลับโหมดคลื่นเป็น "ขาลง/กลับตัว (A-B-C)" รอให้ราคาเด้งทำคลื่น Wave B (แท่งสีฟ้า Aqua ของ CDC ชนแนวต้าน EMA26 ไม่ผ่าน) ค่อยดักเข้า Sell / Short กินคลื่น C หรือรอเกิด Regular Bullish Divergence ที่ก้นคลื่นใหม่ค่อยกลับเข้า Long

กรณีที่ 3: วิ่งใกล้เป้าแต่เกิด Bearish Divergence + แท่งหางยาวบนยอด Plan C (Win-Win)

• อาการบนกราฟ: ราคาวิ่งขึ้นไปโซนกำไร (80,824.5 USDT) ยังไม่ทันถึงเป้า TP2 (82,483.9) แต่เกิดสัญญาณเตือนล่วงหน้าคือ Regular Bearish Divergence ใน RSI + MACD Histogram หดตัวลง + เกิดแท่งเทียนไส้บนยาว (Shooting Star / Pin Bar)
• วิธีรับมือแบบมืออาชีพ: ไม่ต้องเกี่ยงให้ถึงราคาเป๊ะๆ ให้ แบ่งขายทำกำไร (Take Profit) ออกมาก่อน 50% – 70% ทันที แล้ว ยกเส้น Stop Loss ขึ้นมาล็อกกำไรไว้ใต้เส้น CDC Slow EMA(26) ล่าสุด (Trailing Stop)
• ผลลัพธ์: ไม้นี้ของคุณจะกลายเป็น "ชนะ 100% (Risk-Free Win)" ไม่ว่ากราฟจะพุ่งต่อหรือทุบลง คุณก็ได้กำไรเข้าพอร์ตแน่นอน!
