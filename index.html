โค้ด Pine Script: CDC ActionZone V3 2020 สำหรับนำไปวางใน TradingView 
//@version=5
indicator("CDC ActionZone V3 2020", overlay=true)
xsrc = input.source(close, "Source")
xprd1 = input.int(12, "Fast EMA Period")
xprd2 = input.int(26, "Slow EMA Period")
xsmooth = input.int(1, "Smoothing Period")

xPrice = ta.ema(xsrc, xsmooth)
FastMA = ta.ema(xPrice, xprd1)
SlowMA = ta.ema(xPrice, xprd2)

Bull = FastMA > SlowMA
Bear = FastMA < SlowMA

Green  = Bull and xPrice > FastMA
Blue   = Bear and xPrice > FastMA and xPrice > SlowMA
LBlue  = Bear and xPrice > FastMA and xPrice < SlowMA
Red    = Bear and xPrice < FastMA
Orange = Bull and xPrice < FastMA and xPrice < SlowMA
Yellow = Bull and xPrice < FastMA and xPrice > SlowMA

bColor = Green ? color.lime : Blue ? color.blue : LBlue ? color.aqua : Red ? color.red : Orange ? color.orange : Yellow ? color.yellow : color.black
barcolor(bColor)
slPct = input.float(3.0, "% Stop Loss")
rrRatio = input.float(2.0, "Risk:Reward Ratio (TP)")
isLong = input.bool(true, "Direction: true=Long, false=Short")

tpPrice = isLong ? close * (1 + (slPct * rrRatio) / 100) : close * (1 - (slPct * rrRatio) / 100)
slPrice = isLong ? close * (1 - slPct / 100) : close * (1 + slPct / 100)

plot(FastMA, "Fast EMA(12)", color=color.red, linewidth=1)
plot(SlowMA, "Slow EMA(26)", color=color.blue, linewidth=2)
plot(tpPrice, "Take Profit Line", color=color.new(color.lime, 20), style=plot.style_linebr, linewidth=2)
plot(slPrice, "Stop Loss Line", color=color.new(color.red, 20), style=plot.style_linebr, linewidth=2)
plotshape(Bull and Bear[1], "Buy (เขียวแรก)", shape.labelup, location.belowbar, color.lime, text="BUY", textcolor=color.black)
plotshape(Bear and Bull[1], "Sell (แดงแรก)", shape.labeldown, location.abovebar, color.red, text="SELL", textcolor=color.white)
